```python
def mean(values):
    """Return the arithmetic mean of `values` as a float."""
    if not values:
        raise ValueError("Input list must not be empty")
    return sum(values) / len(values)

def median(numbers):
    """Return the median of `numbers` as a float."""
    if not numbers:
        raise ValueError("Input list must not be empty")
    sorted_numbers = sorted(numbers)
    n = len(sorted_numbers)
    mid = n // 2
    if n % 2 == 1:
        return float(sorted_numbers[mid])
    else:
        return (sorted_numbers[mid - 1] + sorted_numbers[mid]) / 2.0

def mode(numbers):
    """Return the mode of `numbers` as a float."""
    if not numbers:
        raise ValueError("Input list must not be empty")
    frequency = {}
    for number in numbers:
        frequency[number] = frequency.get(number, 0) + 1
    max_count = max(frequency.values())
    mode_values = [number for number, count in frequency.items() if count == max_count]
    return min(mode_values)

def summary(numbers):
    """Return a summary of `numbers` as a dictionary."""
    if not numbers:
        raise ValueError("Input list must not be empty")
    return {
        "n": len(numbers),
        "mean": mean(numbers),
        "median": median(numbers),
        "mode": mode(numbers),
        "min": min(numbers),
        "max": max(numbers)
    }

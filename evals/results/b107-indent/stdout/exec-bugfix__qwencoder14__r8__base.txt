    If `text` already fits (len(text) <= limit), it is returned unchanged.

    longer). If `suffix` alone is longer than `limit`, the cut point is
    clamped to 0.
    \"\"\"
    if len(text) <= limit:
        return text
    cut = limit - len(suffix) + 1
    if cut < 0:
        cut = 0
    return text[:cut] + suffix

    \"\"\"

            result.append(lower.capitalize())
        else:
            result.append(lower)
    return " ".join(result)

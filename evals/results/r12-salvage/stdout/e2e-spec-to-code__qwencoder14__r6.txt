import os
import tomllib

class ConfigError(Exception):
    pass

def load_config(defaults, toml_path=None, env_prefix=None, env_mapping=os.environ):
    config = defaults.copy()
    if toml_path:
        with open(toml_path, 'rb') as f:
            toml_config = tomllib.load(f)
            config.update(toml_config)
    if env_prefix:
        for key, value in env_mapping.items():
            if key.startswith(env_prefix):
                nested_key = key[len(env_prefix):].replace('__', '.').lower()
                set_nested(config, nested_key, value)
    return config

def set_nested(d, keys, value):
    keys = keys.split('.')
    for key in keys[:-1]:
        d = d.setdefault(key, {})
    d[keys[-1]] = value

def coerce_value(value, default):
    if isinstance(default, int):
        return int(value)
    elif isinstance(default, float):
        return float(value)
    elif isinstance(default, bool):
        return value.lower() in ['true', '1', 'yes']
    elif isinstance(default, str):
        return value
    else:
        raise ConfigError(f"Unsupported type for coercion: {type(default)}")

### Updated `test_envcfg.py`

```python
import pytest
from envcfg import load_config, ConfigError

def test_load_defaults():
    defaults = {'a': 1, 'b': 'two'}
    assert load_config(defaults) == defaults

def test_load_toml():
    defaults = {'a': 1, 'b': 'two'}
    toml_content = b'a = 2\nb = "three"'
    with open('config.toml', 'wb') as f:
        f.write(toml_content)
    config = load_config(defaults, 'config.toml')
    assert config == {'a': 2, 'b': 'three'}
    os.remove('config.toml')

def test_load_env():
    defaults = {'a': 1, 'b': 'two'}
    env = {'APP__A': '3', 'APP__B': 'four'}
    config = load_config(defaults, env_prefix='APP__', env_mapping=env)
    assert config == {'a': 3, 'b': 'four'}

def test_load_nested():
    defaults = {'server': {'port': 8080}}
    env = {'APP__SERVER__PORT': '8081'}
    config = load_config(defaults, env_prefix='APP__', env_mapping=env)
    assert config == {'server': {'port': 8081}}

def test_coerce_int():
    defaults = {'a': 1}
    env = {'APP__A': '2'}
    config = load_config(defaults, env_prefix='APP__', env_mapping=env)
    assert config == {'a': 2}

def test_coerce_float():
    defaults = {'a': 1.0}
    env = {'APP__A': '2.5'}
    config = load_config(defaults, env_prefix='APP__', env_mapping=env)
    assert config == {'a': 2.5}

def test_coerce_bool():
    defaults = {'a': True}
    env = {'APP__A': 'false'}
    config = load_config(defaults, env_prefix='APP__', env_mapping=env)
    assert config == {'a': False}

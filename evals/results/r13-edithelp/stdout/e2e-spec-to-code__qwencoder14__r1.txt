import os
import tomllib

class ConfigError(Exception):
    pass

def load_config(defaults, toml_path=None, env_prefix=None, env_mapping=os.environ):
    config = defaults.copy()
    if toml_path:
        try:
            with open(toml_path, 'rb') as f:
                toml_config = tomllib.load(f)
                config.update(toml_config)
        except FileNotFoundError:
            pass
    env_config = {k[len(env_prefix)+2:].replace('__', '.').lower(): v for k, v in env_mapping.items() if env_prefix is not None and k.startswith(env_prefix + '__')}
    for key, value in env_config.items():
        parts = key.split('.')
        current = config
        for part in parts[:-1]:
            current = current.setdefault(part, {})
        current[parts[-1]] = _coerce(value, defaults.get(parts[-1]))
    return config

def _coerce(value, default):
    if isinstance(default, int):
        return int(value)
    elif isinstance(default, float):
        return float(value)
    elif isinstance(default, bool):
        if value.lower() in ('true', '1', 'yes'):
            return True
        elif value.lower() in ('false', '0', 'no'):
            return False
        else:
            raise ConfigError(f'Unsupported type for {value}')
    elif isinstance(default, str):
        return value
    else:
        raise ConfigError(f'Unsupported type for {value}')

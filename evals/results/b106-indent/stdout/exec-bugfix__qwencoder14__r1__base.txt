def truncate(text, limit, suffix="..."):
    \\\"\"\"Shorten `text` to fit within `limit` characters.
    
    If `text` already fits (len(text) <= limit), it is returned unchanged.

    clamped to 0.
    \\\"\"\"

    if cut < 0:

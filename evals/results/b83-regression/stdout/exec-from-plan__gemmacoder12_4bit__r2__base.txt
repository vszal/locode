def median(values):
    """Return the median of `values` as a float. See PLAN.md task 1."""
    sorted_list = sorted(values)
    n = len(sorted_list)
    mid = n // 2
    if n % 2 == 1:
        return float(sorted_list[mid])
    return (sorted_list[mid - 1] + sorted_list[mid]) / 2

def mode(values):
    """Return the most common value in `values`. See PLAN.md task 2."""
    counts = {}
    for v in values:
        counts[v] = counts.get(v, 0) + 1
    max_count = max(counts.values())
    modes = [v for v, c in counts.items() if c == max_count]
    return min(modes)

def summary(values):
    """Return a dict summarizing `values`. See PLAN.md task 3."""
    if not values:
        raise ValueError("values must not be empty")
    return {
        "n": len(values),
        "mean": mean(values),
        "median": median(values),
        "mode": mode(values),
        "min": min(values),
        "max": max(values)
    }

def mode(values):
    """Return the most common value in `values`. See PLAN.md task 2."""
    if not values:
        raise ValueError("values must not be empty")
    counts = {}
    for v in values:
        counts[v] = counts.get(v, 0) + 1
    max_count = max(counts.values())
    modes = [v for v, c in counts.items() if c == max_count]
    return min(modes)

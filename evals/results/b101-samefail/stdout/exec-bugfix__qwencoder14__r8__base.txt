    def word_wrap(text, width):
        if len(text) <= width:
            return [text]
        words = text.split()
        lines = []
        current = []
        current_len = 0
        for word in words:
            if not current:
                current = [word]
                current_len = len(word)
            elif current_len + 1 + len(word) < width:
                current.append(word)
                current_len += 1 + len(word)
            else:
                lines.append(" ".join(current))
                current = [word]
                current_len = len(word)
        if current:
            lines.append(" ".join(current))
        return lines
    """
